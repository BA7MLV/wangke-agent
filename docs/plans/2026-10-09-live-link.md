```ts
// src/livelink/protocol.ts
/**
 * 实时链路的帧编解码与分片。
 *
 * 为什么二进制帧自带 12 字节头（rid + seq）而不是「先发一个文本头再发裸字节」：
 * 窗口允许多个 range 在途（host 侧并发读 OPFS），顺序到达不保证，「文本头 + 裸字节」
 * 的约定会在乱序时张冠李戴。rid 关联请求，seq 对视频是字节偏移、对单元是分片序号——
 * 两者都是「接收端据此拼装」的定位量，复用同一个头。
 *
 * 坏帧一律返回 null 不抛：远端数据不完全可信，一条畸形帧不该杀死整条链路
 * （不变量见设计文档 §2）。
 */
import { gzipSync, gunzipSync, strToU8, strFromU8 } from 'fflate';
import { PART_BYTES } from '../sync/units';

export const PROTOCOL_VERSION = 1;
export const BINARY_HEADER_BYTES = 12;

export interface Frame {
  v: number;
  t: string;
  id: string;
  [k: string]: unknown;
}

export function encodeFrame(frame: Frame): string {
  return JSON.stringify(frame);
}

/** 解析失败或版本不符返回 null；调用方计入跳过数，不该让整轮同步炸掉 */
export function decodeFrame(text: string): Frame | null {
  if (typeof text !== 'string' || text.length === 0) return null;
  let obj: unknown;
  try {
    obj = JSON.parse(text);
  } catch {
    return null;
  }
  if (obj == null || typeof obj !== 'object' || Array.isArray(obj)) return null;
  const f = obj as Partial<Frame>;
  if (f.v !== PROTOCOL_VERSION) return null;
  if (typeof f.t !== 'string' || typeof f.id !== 'string') return null;
  return f as Frame;
}

/** 二进制帧：4B rid（LE u32）+ 8B seq（LE f64）+ payload */
export function encodeBinaryFrame(rid: number, seq: number, payload: Uint8Array): ArrayBuffer {
  const buf = new ArrayBuffer(BINARY_HEADER_BYTES + payload.byteLength);
  const view = new DataView(buf);
  view.setUint32(0, rid, true);
  view.setFloat64(4, seq, true);
  new Uint8Array(buf, BINARY_HEADER_BYTES).set(payload);
  return buf;
}

export function decodeBinaryFrame(buf: ArrayBuffer): { rid: number; seq: number; payload: Uint8Array } | null {
  if (buf.byteLength < BINARY_HEADER_BYTES) return null;
  const view = new DataView(buf);
  return {
    rid: view.getUint32(0, true),
    seq: view.getFloat64(4, true),
    payload: new Uint8Array(buf, BINARY_HEADER_BYTES),
  };
}

/** 压缩 + 分片。空载荷也返回 1 片（见 test 5 的理由） */
export function chunkUnit(payload: Uint8Array): Uint8Array[] {
  const parts: Uint8Array[] = [];
  for (let off = 0; off < Math.max(payload.byteLength, 1); off += PART_BYTES) {
    parts.push(payload.subarray(off, Math.min(payload.byteLength, off + PART_BYTES)));
    if (payload.byteLength === 0) break;
  }
  return parts;
}

/** 按 seq 排序拼接（分片可能乱序到达） */
export function joinUnit(parts: Uint8Array[]): Uint8Array {
  const sorted = [...parts].sort((a, b) => 0); // 调用方已按 seq 对齐顺序；此处仅占位保持纯函数可测
  const total = sorted.reduce((n, p) => n + p.byteLength, 0);
  const out = new Uint8Array(total);
  let off = 0;
  for (const p of sorted) {
    out.set(p, off);
    off += p.byteLength;
  }
  return out;
}

/** 单元载荷：JSON → gzip。gzip 与分片阈值都直接复用云同步设计里的现成件 */
export function encodeUnitPayload(json: string): Uint8Array {
  return gzipSync(strToU8(json));
}

export function decodeUnitPayload(bytes: Uint8Array): string {
  return strFromU8(gunzipSync(bytes));
}
```

> ⚠️ 实现时修正 `joinUnit`：上面 sort 占位是错的——应改为按 seq 外部排好序后传入，函数只做拼接。若 Test 6 因此红，把测试里的 parts 改为已排序数组，函数签名保持 `joinUnit(parts: Uint8Array[]): Uint8Array` 纯拼接。

### Task 2: 拉流窗口计算（window.ts）

> **变更（2026-10-10，Task 2 已完成并修正）**：`PlanOpts.seek` 已改为 `pos`（播放头所在字节偏移）。
> 原因：窗口锚在「当前连续覆盖末端」会追着末端跑（无界），锚在「文件头」则首窗填满后
> `end < stop` 恒不成立、每轮返回空——8MB 窗口约 13 秒，播到 13 秒永久停摆，且 MSE
> 淘汰已播缓冲也救不了写死的锚点。只有锚在播放头（`stop = min(size, pos + win)`）才能
> 「边播边拉」：播放推进 → 锚点推进 → 窗口滑动；seek 就是 pos 跳变，自然在新位置开窗，
> 且不回头补播放头之后的数据。**本节与 Task 6 的代码以 `src/livelink/window.ts` 和
> `scripts/test-livelink-window.mjs` 的最终实现为准**（下面 Step 1/3 的参考代码是初版，
> 保留仅供对照，勿照抄）。

```ts
// src/livelink/window.ts
/**
 * 「下一轮该拉哪些字节段」的纯计算。
 *
 * 语义（与设计文档 §3.3 视频面一致）：
 *   - windowBytes = 从**当前连续覆盖末端**起算，愿意提前缓冲的总量。每一轮重算，
 *     播放推进 → 末端推进 → 窗口滑动，表现就是「边播边拉」。
 *   - seek 优先：seek 目标不在缓冲里时先拉 seek 那片（用户此刻就要看），
 *     再把连续覆盖到 seek 之前的洞补上——两者都不得与已缓冲/在途重叠
 *     （MSE 重复 append 同一段字节会直接损坏已解码帧）。
 *   - 在途（inflight）占满配额就不规划新请求：背压从这里来，不从 socket 缓冲来。
 */
export interface Range { start: number; end: number } // 字节，end 开区间
export interface Fetch { start: number; len: number }

export interface PlanOpts {
  size: number;
  buffered: Range[];
  chunk: number;
  windowBytes: number;
  inflight: number;
  maxInflight: number;
  seek?: number | null;
}

export function planFetches(o: PlanOpts): Fetch[] {
  if (!Number.isFinite(o.size) || o.size <= 0 || !Number.isFinite(o.chunk) || o.chunk <= 0) return [];
  if (!Number.isFinite(o.inflight) || !Number.isFinite(o.maxInflight)) return [];

  // pending 同时充当「已缓冲 + 本轮已规划」的覆盖表，规划即写入，天然防重叠
  const pending: Range[] = o.buffered
    .filter((r) => r.end > r.start)
    .map((r) => ({ start: Math.max(0, r.start), end: Math.min(o.size, r.end) }))
    .sort((a, b) => a.start - b.start);
  const out: Fetch[] = [];
  let capacity = Math.max(0, Math.floor(o.maxInflight) - Math.floor(o.inflight));

  const add = (start: number, len: number) => {
    if (capacity <= 0 || len <= 0) return;
    const s = Math.max(0, Math.min(o.size, start));
    const e = Math.min(o.size, s + len);
    if (e <= s) return;
    out.push({ start: s, len: e - s });
    pending.push({ start: s, end: e });
    pending.sort((a, b) => a.start - b.start);
    capacity--;
  };

  if (o.seek != null && Number.isFinite(o.seek)) add(o.seek, o.chunk);

  // 连续覆盖末端：从 0 起跳，能被首尾相接的区间一路推到哪里
  const contigEnd = (): number => {
    let cursor = 0;
    for (;;) {
      const hit = pending.find((r) => r.start <= cursor && cursor < r.end);
      if (!hit) return cursor;
      cursor = hit.end;
    }
  };

  const stop = Math.min(o.size, contigEnd() + Math.max(0, o.windowBytes));
  while (capacity > 0) {
    const end = contigEnd();
    if (end >= stop) break;
    const next = pending.filter((r) => r.start >= end).sort((a, b) => a.start - b.start)[0];
    const limit = Math.min(stop, next ? next.start : o.size);
    const before = out.length;
    add(end, limit - end);
    if (out.length === before) break; // add 没成功（容量耗尽），防死循环
  }
  return out;
}
```

```ts
// src/livelink/payloads.ts
/**
 * host 侧「把 DB 行变成链路载荷」的纯函数集合。
 *
 * 单元 id 直接复用 sync/units.ts 的 videoUnit()：同一门课在两套系统里必须是同一个 id，
 * 两种切法必然漂移（不变量 8）。P1 只发 meta / content / cover 三种；
 * frames（讲义插图）不拉，讲义 IR 内引用处由 guest 显示占位。
 *
 * 字段取舍：VIDEO_META_FIELDS 里剔除本机事实三项（fileDeleted 例外，见下），
 * 因为 guest 镜像只进内存，不需要 replicas 意义上的权威性。
 */
import type { VideoRow, SegmentRow, SubtitleTrackRow, CoverRow } from '../store/db';
import { videoUnit } from '../sync/units';

export interface IndexEntry {
  id: string;
  name: string;
  kind: 'video' | 'material';
  duration: number;
  size: number;
  mimeType: string;
  createdAt: number;
  status: string;
  materialFormat?: string;
  /** 文件本体已删 → guest 侧 openFile 必失败，索引要让它能提前说明原因 */
  fileDeleted: boolean;
}

export function buildIndex(rows: VideoRow[]): IndexEntry[] {
  return rows.map((r) => ({
    id: r.id,
    name: r.name,
    kind: r.kind ?? 'video',
    duration: r.duration,
    size: r.size,
    mimeType: r.mimeType,
    createdAt: r.createdAt,
    status: r.status,
    materialFormat: r.materialFormat,
    fileDeleted: r.fileDeleted === 1,
  }));
}

export function buildMetaUnit(row: VideoRow): string {
  return JSON.stringify({
    id: row.id,
    name: row.name,
    size: row.size,
    mimeType: row.mimeType,
    duration: row.duration,
    createdAt: row.createdAt,
    status: row.status,
    kind: row.kind,
    materialFormat: row.materialFormat,
    unitCount: row.unitCount,
    lastPosition: row.lastPosition ?? 0,
    finished: row.finished ?? 0,
    folderId: row.folderId,
    dominantColor: row.dominantColor,
    coverState: row.coverState,
    scanned: row.scanned === 1,
    empty: row.empty === 1,
  });
}

export function buildContentUnit(videoId: string, segments: SegmentRow[], tracks: SubtitleTrackRow[]): string {
  return JSON.stringify({
    videoId,
    // 只发显示层要用的：cues 用于句级跳转，status 过滤未转写段
    segments: segments.map((s) => ({ idx: s.idx, start: s.start, end: s.end, text: s.text, cues: s.cues })),
    tracks: tracks.map((t) => ({ lang: t.lang, lanDoc: t.lanDoc, primary: t.primary === 1, cues: t.cues })),
  });
}

export function buildCoverUnit(cover: CoverRow): string {
  return JSON.stringify({
    videoId: cover.videoId,
    w: cover.w,
    h: cover.h,
    ts: cover.ts,
    blobBase64: bytesToBase64(new Uint8Array()), // 占位，真实 host 侧用 blobToBase64（见下）
  });
}

/** host 侧专用：Blob → base64（浏览器）。Node 单测只覆盖纯字段，不碰这条路径 */
export async function coverUnitFromBlob(cover: CoverRow): Promise<string> {
  const buf = new Uint8Array(await cover.blob.arrayBuffer());
  return JSON.stringify({
    videoId: cover.videoId,
    w: cover.w,
    h: cover.h,
    ts: cover.ts,
    blobBase64: bytesToBase64(buf),
  });
}

/** 分片进制避免大图 spread 爆栈（封面 480px 通常 <40KB，但材料封面可能更宽） */
export function bytesToBase64(bytes: Uint8Array): string {
  let bin = '';
  for (let i = 0; i < bytes.length; i += 0x8000) {
    bin += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(bin);
```

```ts
// src/livelink/host.ts
/**
 * host 侧链路客户端：host 页面就是服务器。
 *
 * 三个设计约束落在代码里：
 *   1. 凭据边界——本模块只读 videos/segments/subtitleTracks/covers/files，
 *      永远不读 providers / settings 里的 Key 与 Cookie（不变量 1）。
 *   2. range 读并发上限 2：OPFS slice 是磁盘 IO，家和云之间带宽通常先打满，
 *      更多并发只会让在途字节堆在 socket 里（背压由 window.ts 规划）。
 *   3. 容器白名单在 openFile 时就拒绝：播到一半才发现播不了，比打开时明说更糟。
 */
import { db } from '../store/db';
import { getCourseFile, readCourseFileRange } from '../store/fileStore';
import {
  encodeFrame, decodeFrame, encodeBinaryFrame, decodeBinaryFrame,
  encodeUnitPayload, chunkUnit,
  type Frame,
} from './protocol';
import { buildIndex, buildMetaUnit, buildContentUnit, coverUnitFromBlob } from './payloads';

const MAX_RANGE_INFLIGHT = 2;
const RANGE_CHUNK = 1024 * 1024; // 1MB/帧：与窗口 8MB 匹配，seek 粒度也够细
const DIRECT_CONTAINERS = ['video/mp4', 'video/webm'];

export interface HostEvents {
  onGuests(guests: { id: string }[]): void;
  onStatus(s: 'connecting' | 'online' | 'off'): void;
  onError(msg: string): void;
}

export class LiveHost {
  private ws: WebSocket | null = null;
  private guests = new Map<string, { id: string }>();
  private rangeInflight = 0;
  private stopped = false;

  constructor(
    private opts: { relayUrl: string; code: string; events: HostEvents },
  ) {}

  start(): void {
    this.stopped = false;
    this.opts.events.onStatus('connecting');
    const url = `${this.opts.relayUrl}?room=${encodeURIComponent(this.opts.code)}&role=host`;
    const ws = new WebSocket(url);
    this.ws = ws;
    ws.onopen = () => this.opts.events.onStatus('online');
    ws.onclose = () => {
      this.ws = null;
      this.guests.clear();
      this.opts.events.onGuests([]);
      if (!this.stopped) this.opts.events.onStatus('off');
    };
    ws.onerror = () => this.opts.events.onError('中继连接失败，检查中继地址');
    ws.onmessage = (e) => this.handle(e.data);
  }

  stop(): void {
    this.stopped = true;
    this.ws?.close();
    this.ws = null;
    this.guests.clear();
    this.opts.events.onGuests([]);
    this.opts.events.onStatus('off');
  }

  private send(frame: Frame): void {
    if (this.ws?.readyState === WebSocket.OPEN) this.ws.send(encodeFrame(frame));
  }

  /** 新 rid：同一秒内两次调用也要不同（回放 ack 靠它匹配） */
  private rid(): number {
    return (Math.random() * 0xffffffff) >>> 0;
  }

  private async handle(data: unknown): Promise<void> {
    if (typeof data === 'string') {
      const f = decodeFrame(data);
      if (!f) return; // 坏帧跳过，不杀链路
      if (f.t === 'hello') {
        const gid = String(f.id);
        this.guests.set(gid, { id: gid });
        this.opts.events.onGuests([...this.guests.values()]);
        this.send({ v: 1, t: 'welcome', id: gid });
        return;
      }
      if (f.t === 'bye') {
        this.guests.delete(String(f.id));
        this.opts.events.onGuests([...this.guests.values()]);
        return;
      }
      if (f.t === 'index.req') {
        const rows = await db.videos.orderBy('createdAt').toArray();
        this.send({ v: 1, t: 'index.res', id: f.id, courses: buildIndex(rows) });
        return;
      }
      if (f.t === 'unit.req') {
        await this.handleUnitReq(f);
        return;
      }
      if (f.t === 'video.open') {
        await this.handleVideoOpen(f);
        return;
      }
      if (f.t === 'video.range') {
        await this.handleVideoRange(f);
        return;
      }
      return;
    }
    if (data instanceof ArrayBuffer) {
      const b = decodeBinaryFrame(data);
      if (!b) return;
      // host 不收二进制（写面 op 是 P2 的事）
      return;
    }
  }

  private async handleUnitReq(f: Frame): Promise<void> {
    const unit = String(f.unit);
    const rid = this.rid();
    if (unit.endsWith(':meta')) {
      const row = await db.videos.get(unit.slice('video:'.length, -':meta'.length));
      if (!row) {
        this.send({ v: 1, t: 'unit.err', id: f.id, reason: 'not-found' });
        return;
      }
      this.sendUnit(f.id as string, rid, buildMetaUnit(row));
      return;
    }
    if (unit.endsWith(':content')) {
      const videoId = unit.slice('video:'.length, -':content'.length);
      const [segments, tracks] = await Promise.all([
        db.segments.where('videoId').equals(videoId).toArray(),
        db.subtitleTracks.where('videoId').equals(videoId).toArray(),
      ]);
      this.sendUnit(f.id as string, rid, buildContentUnit(videoId, segments, tracks));
      return;
    }
    if (unit.endsWith(':cover')) {
      const cover = await db.covers.get(unit.slice('video:'.length, -':cover'.length));
      if (!cover) {
        this.send({ v: 1, t: 'unit.err', id: f.id, reason: 'not-found' });
        return;
      }
      const json = await coverUnitFromBlob(cover);
      this.sendUnit(f.id as string, rid, json);
      return;
    }
    this.send({ v: 1, t: 'unit.err', id: f.id, reason: 'unsupported-unit' });
  }

  /** 先发 res 头（分片数），再按序发二进制分片。rid 串起两者 */
  private sendUnit(reqId: string, rid: number, json: string): void {
    const bytes = encodeUnitPayload(json);
    const parts = chunkUnit(bytes);
    this.send({ v: 1, t: 'unit.res', id: reqId, parts: parts.length, len: bytes.byteLength });
    parts.forEach((p, i) => this.ws?.send(encodeBinaryFrame(rid, i, p)));
  }

  private async handleVideoOpen(f: Frame): Promise<void> {
    const videoId = String(f.videoId);
    const row = await db.videos.get(videoId);
    if (!row) {
      this.send({ v: 1, t: 'video.meta', id: f.id, ok: false, reason: 'not-found' });
      return;
    }
    if (row.fileDeleted === 1) {
      this.send({ v: 1, t: 'video.meta', id: f.id, ok: false, reason: 'file-deleted' });
      return;
    }
    if (!DIRECT_CONTAINERS.includes(row.mimeType)) {
      this.send({ v: 1, t: 'video.meta', id: f.id, ok: false, reason: 'unsupported-container', mime: row.mimeType });
      return;
    }
    const file = await getCourseFile(videoId);
    if (!file) {
      this.send({ v: 1, t: 'video.meta', id: f.id, ok: false, reason: 'file-missing' });
      return;
    }
    this.send({ v: 1, t: 'video.meta', id: f.id, ok: true, size: file.size, mime: row.mimeType });
  }

  private async handleVideoRange(f: Frame): Promise<void> {
    if (this.rangeInflight >= MAX_RANGE_INFLIGHT) return; // 背压：直接丢弃，guest 下轮重规划
    this.rangeInflight++;
    try {
      const videoId = String(f.videoId);
      const start = Number(f.start) || 0;
      const len = Math.min(Number(f.len) || RANGE_CHUNK, RANGE_CHUNK);
      const buf = await readCourseFileRange(videoId, start, len);
      if (buf) {
        const payload = new Uint8Array(buf);
        // 大段按 RANGE_CHUNK 再切，单帧不超过 1MB
        for (let off = 0; off < payload.byteLength || off === 0; off += RANGE_CHUNK) {
          const piece = payload.subarray(off, Math.min(payload.byteLength, off + RANGE_CHUNK));
          this.ws?.send(encodeBinaryFrame(Number(f.id) >>> 0, start + off, piece));
          if (payload.byteLength === 0) break;
        }
      }
    } finally {
      this.rangeInflight--;
    }
  }
}
```

```ts
  enabled: boolean;

/** 8 位 base32 风格配对码：去掉易混淆字符（0/O/1/I），朗读与手输都友好 */
function newCode(): string {
  const alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  const bytes = new Uint8Array(8);
  crypto.getRandomValues(bytes);
  return [...bytes].map((b) => alphabet[b % alphabet.length]).join('');
}

      enabled: false,
      code: '',
      guests: [],
      status: 'off',
      error: '',

        const relay = useSettings.getState().liveRelay.trim();
        if (!relay) {
          set({ error: '先填中继地址（部署 live-relay 后得到）' });
          return;

        host?.stop();
        host = new LiveHost({
          relayUrl: relay,
          code,

            onGuests: (guests) => set({ guests }),
            onStatus: (status) => set({ status, error: status === 'online' ? '' : get().error }),
            onError: (msg) => set({ status: 'error', error: msg }),
          },
        });
        host.start();
        set({ enabled: true, code, error: '' });
      },
      disable: () => {
        host?.stop();

        set({ enabled: false, guests: [], status: 'off' });
      },
      rotate: () => {
        const next = newCode();
        get().disable();
        set({ code: next });
        get().enable();
      },
    }),
    {
      name: 'wangke-livelink',
      partialize: (s) => ({ code: s.code }) as unknown as LiveLinkState,
```

```ts
// src/livelink/guest.ts
/**
 * guest 侧链路客户端。
 *
 * 两条铁律落在实现里：
 *   1. 拉到的数据只进内存（不变量 9）——落库会产生「iPad 上半个课程库」的脏状态，
 *      并让人误以为 guest 是独立副本（那是云同步的场景）。
 *   2. LLM 调用永远不在本模块——问答属 P2，且必须走 guest 自己的 Key（不变量 1）。
 */
import {
  encodeFrame, decodeFrame, decodeBinaryFrame,
  type Frame,
} from './protocol';

export interface IndexEntry { id: string; name: string; kind: string; duration: number; mimeType: string; fileDeleted: boolean }
export interface VideoMetaOk { ok: true; size: number; mime: string }
export interface VideoMetaErr { ok: false; reason: string; mime?: string }

interface PendingUnit {
  parts: Map<number, Uint8Array>;
  expect: number;
  resolve: (u: Uint8Array) => void;
  reject: (e: Error) => void;
}

export class LiveGuest {
  private ws: WebSocket | null = null;
  private units = new Map<string, PendingUnit>();
  private pending = new Map<string, { resolve: (f: Frame) => void; reject: (e: Error) => void }>();
  private ridSeq = 1;

  constructor(
    private opts: {
      relayUrl: string;
      code: string;
      onBinaryFrame?: (rid: number, seq: number, payload: Uint8Array) => void;
      onHostDown?: () => void;
      onHostUp?: () => void;
    },
  ) {}

  connect(): Promise<void> {
    return new Promise((resolve, reject) => {
      const url = `${this.opts.relayUrl}?room=${encodeURIComponent(this.opts.code)}&role=guest`;
      const ws = new WebSocket(url);
      this.ws = ws;
      ws.onopen = () => {
        this.send({ v: 1, t: 'hello', id: 'guest' });
        resolve();
      };
      ws.onerror = () => reject(new Error('中继连接失败'));
      ws.onmessage = (e) => this.handle(e.data);
      ws.onclose = () => {
        if (!this.opts.onHostDown) return;
        this.opts.onHostDown();
      };
    });
  }

  close(): void {
    this.send({ v: 1, t: 'bye', id: 'guest' });
    this.ws?.close();
    this.ws = null;
  }

  private send(f: Frame): void {
    if (this.ws?.readyState === WebSocket.OPEN) this.ws.send(encodeFrame(f));
  }

  private nextId(): string {
    return `r${this.ridSeq++}-${Math.random().toString(36).slice(2, 8)}`;
  }

  private request(frame: Omit<Frame, 'id'>): Promise<Frame> {
    const id = this.nextId();
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject });
      this.send({ ...frame, id } as Frame);
      window.setTimeout(() => {
        if (this.pending.delete(id)) reject(new Error('请求超时'));
      }, 15000);
    });
  }

  private handle(data: unknown): void {
    if (typeof data === 'string') {
      const f = decodeFrame(data);
      if (!f) return;
      if (f.t === 'unit.res') {
        const p = this.units.get(f.id);
        if (p && !p.expect) p.expect = Number(f.parts);
        return;
      }
      if (f.t === 'unit.err') {
        this.pending.get(f.id)?.reject(new Error(String(f.reason)));
        this.pending.delete(f.id);
        return;
      }
      if (f.t === 'host-link-down') {
        this.opts.onHostDown?.();
        return;
      }
      if (f.t === 'host-link-up') {
        this.opts.onHostUp?.();
        return;
      }
      const p = this.pending.get(f.id);
      if (p) {
        this.pending.delete(f.id);
        p.resolve(f);
      }
      return;
    }
    if (data instanceof ArrayBuffer) {
      const b = decodeBinaryFrame(data);
      if (!b) return;
      const p = this.units.get(String(b.rid));
      if (p) {
        p.parts.set(b.seq, b.payload);
        if (p.parts.size >= p.expect) {
          const total = [...p.parts.entries()].sort((a, c) => a[0] - c[0]).reduce((n, [, u]) => n + u.byteLength, 0);
          const out = new Uint8Array(total);
          let off = 0;
          for (const [, u] of [...p.parts.entries()].sort((a, c) => a[0] - c[0])) {
            out.set(u, off);
            off += u.byteLength;
          }
          this.units.delete(String(b.rid));
          p.resolve(out);
        }
        return;
      }
      // 不是单元分片 → 视频帧（mse 消费）
      this.opts.onBinaryFrame?.(b.rid, b.seq, b.payload);
    }
  }

  async loadIndex(): Promise<IndexEntry[]> {
    const res = await this.request({ v: 1, t: 'index.req' });
    return (res.courses ?? []) as IndexEntry[];
  }

  /** 拉一个单元（gzip 拼装 + 解压 + JSON.parse）。调用方拿到即用，不落库 */
  async loadUnit(unit: string): Promise<unknown> {
    const reqId = this.nextId();
    const done = new Promise<Uint8Array>((resolve, reject) => {
      this.units.set(reqId, { parts: new Map(), expect: 0, resolve, reject });
      window.setTimeout(() => {
        if (this.units.delete(reqId)) reject(new Error('单元拉取超时'));
      }, 30000);
    });
    this.send({ v: 1, t: 'unit.req', id: reqId, unit });
    const bytes = await done;
    const { decodeUnitPayload } = await import('./protocol');
    return JSON.parse(decodeUnitPayload(bytes));
  }

  async openVideo(videoId: string): Promise<VideoMetaOk | VideoMetaErr> {
    const res = await this.request({ v: 1, t: 'video.open', videoId });
    return res.ok
      ? { ok: true, size: Number(res.size), mime: String(res.mime) }
      : { ok: false, reason: String(res.reason), mime: res.mime ? String(res.mime) : undefined };
  }
}
```

> ⚠️ **实现注意**（Task 1 已定契约）：`loadUnit` 里 `JSON.parse(decodeUnitPayload(bytes))` 必须包 try/catch —— `decodeUnitPayload` 遇到坏 gzip 会抛（fflate 的契约，protocol.ts 不替它吞，没有 null 通道）。要让 `loadUnit` 的 Promise 直接 reject；漏在外面的后果是错误表现为 30s 超时才暴露，或者异常钻进 socket 事件回调把整条链路打死。

```ts
// src/livelink/mse.ts
/**
 * guest 侧 MSE 播放管道。
 *
 * 关键取舍：拉哪些字节完全由 window.ts 规划，MediaSource.buffered 回喂——
 * 即「播放器真实缓冲状态 → 下一轮 range 请求」，不另起一套预测逻辑。
 * append 用 'sequence' 模式外加强制等待 updateend：并发 appendBuffer 在
 * Safari 上会把缓冲打花，串行是唯一稳的姿势。
 */
import { planFetches, type Range } from './window';

const WINDOW_BYTES = 8 * 1024 * 1024; // 与 host 侧 readCourseFileRange 的语义匹配
const MAX_INFLIGHT = 4;

export interface MseIo {
  openVideo(videoId: string): Promise<{ ok: true; size: number; mime: string } | { ok: false; reason: string }>;
  onRange(rid: number, seq: number, payload: Uint8Array): void;
}

export class GuestMsePlayer {
  private ms: MediaSource | null = null;
  private sb: SourceBuffer | null = null;
  private queue: Uint8Array[] = [];
  private appending = false;
  private inflight = 0;
  private meta: { size: number; mime: string } | null = null;
  private disposed = false;

  constructor(
    private video: HTMLVideoElement,
    private io: MseIo,
    private videoId: string,
  ) {}

  async start(): Promise<void> {
    const meta = await this.io.openVideo(this.videoId);
    if (!meta.ok) throw new Error(`无法播放：${meta.reason}`);
    this.meta = meta;
    const ms = new MediaSource();
    this.ms = ms;
    this.video.src = URL.createObjectURL(ms);
    await new Promise<void>((resolve) => {
      ms.addEventListener('sourceopen', () => resolve(), { once: true });
    });
    this.sb = ms.addSourceBuffer(msCodec(meta.mime));
    this.video.addEventListener('seeked', () => this.fill());
    this.video.addEventListener('timeupdate', () => this.fill());
    this.fill();
  }

  bufferedRanges(): Range[] {
    const out: Range[] = [];
    const b = this.video.buffered;
    for (let i = 0; i < b.length; i++) out.push({ start: b.start(i), end: b.end(i) });
    return out;
  }

  /** 播放位置（秒）→ 字节偏移粗估：host 不知道码率，只能按「已播/总长 × size」线性估，
   *  误差由 seeked 后按真实 seek 字节位重规划兜底（fill(seek) 传视频 currentTime 比例）。 */
  private currentByte(): number | null {
    if (!this.meta || !this.video.duration || !Number.isFinite(this.video.duration)) return null;
    return Math.floor((this.video.currentTime / this.video.duration) * this.meta.size);
  }

  fill(): void {
    if (this.disposed || !this.sb || !this.meta) return;
    // pos = 播放头字节偏移（seek 进行中 video.currentTime 已返回目标位置，无需特判 seeking）
    const plans = planFetches({
      size: this.meta.size,
      buffered: this.bufferedRanges(),
      chunk: 1024 * 1024,
      windowBytes: WINDOW_BYTES,
      inflight: this.inflight,
      maxInflight: MAX_INFLIGHT,
      pos: this.currentByte(),
    });
    for (const p of plans) {
      this.inflight++;
      const rid = (Math.random() * 0xffffffff) >>> 0;
      pendingRanges.set(rid, { start: p.start, len: p.len, player: this });
      this.io.onRange(rid, p.start, p.len);
    }
  }

  dispose(): void {
    this.disposed = true;
    if (this.ms?.readyState === 'open') this.ms.endOfStream?.();
    this.ms = null;
    this.sb = null;
  }
}

/** host mime → MSE codec 串。mp4/webm 容器级 type 即可（codecs 可选），
 *  真解码不了时 appendBuffer 会抛错，由页面 error 态接住。 */
function msCodec(mime: string): string {
  return mime === 'video/webm' ? 'video/webm' : 'video/mp4';
}

/** rid → 分片重组现场。GuestMsePlayer 与 io.onRange 之间的 glue。 */
interface RangeCtx { start: number; len: number; player: GuestMsePlayer }
export const pendingRanges = new Map<number, RangeCtx>();

/** mse.ts 与 guest 页之间的组装入口：host 分片到达 → 追加到对应 range 的现场 */
export function feedRange(rid: number, seq: number, payload: Uint8Array): void {
  const ctx = pendingRanges.get(rid);
  if (!ctx) return;
  ctx.player.appendChunk(payload);
  if (seq + payload.byteLength >= ctx.start + ctx.len) {
    pendingRanges.delete(rid);
    ctx.player.onRangeDone();
  }
}
```

> ⚠️ `appendChunk` / `onRangeDone` 是 `GuestMsePlayer` 的私有方法，上面用 `feedRange` 跨越模块边界触碰它们——实现时把这两个方法改为 public（或在 mse.ts 内部维护 append 队列，不让外部类进入）。选定一种后同步改 `feedRange`。`pendingRanges` 全局 map 同理会引争议，可接受：进程内唯一播放器，P1 不为多播放器实例抽象。

```ts
// cloudflare-worker/live-relay/src/index.ts
/**
 * 实时链文中继：按配对码把 host 与 guest 接进同一房间，转发帧，不解析业务数据。
 *
 * 与 bili-proxy 同一条原则：worker 只搬运。三个细节：
 *   1. WebSocket Hibernation API——长连接空闲时不烧 CPU，平台派发 message/close/alarm；
 *   2. host 断开 → 广播 host-link-down + 60s alarm 宽限；host 重连则取消 alarm，
 *      guest 侧收到 host-link-up 自动恢复（设计文档不变量 7）；
 *   3. 错误配对限速：guest 连入而房间无 host = 配对码不存在，计入 storage，
 *      满 5 次锁 5 分钟（设计文档 §3.6）。
 */
import { DurableObject } from 'cloudflare:workers';

export { Room };

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    if (request.headers.get('Upgrade') !== 'websocket') {
      return new Response('expected websocket', { status: 426 });
    }
    const room = url.searchParams.get('room')?.trim();
    const role = url.searchParams.get('role');
    if (!room || (role !== 'host' && role !== 'guest')) {
      return new Response('bad params', { status: 400 });
    }
    const id = env.ROOM.idFromName(room);
    const stub = env.ROOM.get(id);
    return stub.fetch(request);
  },
};

interface Env {
  ROOM: DurableObjectNamespace<Room>;
}

export class Room extends DurableObject {
  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url);
    const role = url.searchParams.get('role')!;
    const pair = new WebSocketPair();
    const [client, server] = Object.values(pair) as [WebSocket, WebSocket];
    this.ctx.acceptWebSocket(server);
    server.serializeAttachment({ role });
    if (role === 'guest') {
      const hasHost = this.ctx.getWebSockets().some((s) => this.role(s) === 'host');
      if (!hasHost) {
        const bad = ((await this.ctx.storage.get<number>('badAttempts')) ?? 0) + 1;
        await this.ctx.storage.put('badAttempts', bad);
        if (bad >= 5) {
          await this.ctx.storage.setAlarm(Date.now() + 5 * 60 * 1000); // 到点清零
          server.close(4403, 'too many attempts');
        } else {
          server.close(4001, 'bad code');
        }
        return new Response(null, status: 101, webSocket: client });
      }
    }
    return new Response(null, { status: 101, webSocket: client });
  }

  private role(ws: WebSocket): string | undefined {
    return ws.deserializeAttachment<{ role: string }>()?.role;
  }

  webSocketMessage(ws: WebSocket, message: string | ArrayBuffer): void {
    const role = this.role(ws);
    if (!role) return;
    for (const other of this.ctx.getWebSockets()) {
      if (other === ws) continue;
      if (this.role(other) === role) continue; // 同角色互不可见（host↔host、guest↔guest）
      other.send(message);
    }
  }

  webSocketClose(ws: WebSocket): void {
    const role = this.role(ws);
    if (role !== 'host') return;
    for (const g of this.ctx.getWebSockets()) {
      if (this.role(g) === 'guest') g.send(JSON.stringify({ v: 1, t: 'host-link-down', id: 'relay' }));
    }
    // 宽限 60s：host 刷新/短暂断网时重连即可取消
    this.ctx.storage.setAlarm(Date.now() + 60_000);
  }

  async alarm(): Promise<void> {
    // 无 host 且到点：尝试重置限速计数；仍无 host 则房间自然回收
    const hasHost = this.ctx.getWebSockets().some((s) => this.role(s) === 'host');
    if (!hasHost) {
      await this.ctx.storage.delete('badAttempts');
      for (const ws of this.ctx.getWebSockets()) ws.close(4000, 'room expired');
    }
  }
}
```

```js
  const url = new URL(req.url, 'http://x');
  const room = url.searchParams.get('room');
  const role = url.searchParams.get('role');
  if (!room || !role) return ws.close(4000);
  if (!rooms.has(room)) rooms.set(room, { host: null, guests: new Set() });
  const r = rooms.get(room);
  if (role === 'host') r.host = ws;
  else r.guests.add(ws);
  ws.on('message', (data, isBinary) => {
    if (isBinary && data.length >= 12) {
      stats.rangeFrames++;
      stats.rangeBytes += data.length - 12;
    }
    const target = role === 'host' ? [...r.guests] : r.host ? [r.host] : [];
    for (const t of target) if (t.readyState === 1) t.send(data, { binary: isBinary });
  });
  ws.on('close', () => {
    if (role === 'host') {
      r.host = null;
      for (const g of r.guests) if (g.readyState === 1) g.send(JSON.stringify({ v: 1, t: 'host-link-down', id: 'relay' }));
    } else r.guests.delete(ws);
  });
});

// ── 播种数据 ──────────────────────────────────────────────────────────────
const VIDEO_ID = 'live-e2e-1';
const SEG_TEXT = '第一段：实时链路从这里开始';
const SUB_BYTES = new Uint8Array(5 * 1024 * 1024).map((_, i) => (i * 7) % 256); // 确定性伪随机
const TEST_FILE = process.env.TEST_FILE;

async function seedHost(page) {
  await page.goto(`${BASE}/#/settings`);
  await page.evaluate(async () => {
    const { db } = await import('/src/store/db.ts');
    const { saveVideoFile } = await import('/src/store/fileStore.ts');
    await db.videos.put({ id: 'live-e2e-1', name: '实时链路测试课', size: 5 * 1024 * 1024, mimeType: 'video/mp4', duration: 60, createdAt: Date.now(), status: 'transcribed', kind: 'video' });
    await db.segments.put({ videoId: 'live-e2e-1', idx: 0, start: 0, end: 5, text: '第一段：实时链路从这里开始', status: 1 });
    await db.subtitleTracks.put({ videoId: 'live-e2e-1', lang: 'ai-en', lanDoc: '英语', primary: 0, cues: [{ start: 0, end: 5, text: 'hello live link' }] });
    const coverBlob = new Blob([new Uint8Array([1, 2, 3, 4])], { type: 'image/webp' });
    await db.covers.put({ videoId: 'live-e2e-1', blob: coverBlob, w: 480, h: 270, ts: 0 });
    await saveVideoFile('live-e2e-1', window.__E2E_BLOB);
  }, { arg: undefined });
  // blob 必须经参数传入（不能序列化 Blob 字面量过 evaluate）
}
```

> `await page.evaluate((b64) => { window.__E2E_BLOB = Uint8Array.from(atob(b64), c => c.charCodeAt(0)); }, fileB64)`
